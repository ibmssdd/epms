import 'package:flutter/material.dart';
import 'package:sqflite_common_ffi/sqflite_ffi.dart';

import 'database/database_helper.dart';

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();

  // Initialize SQLite for Windows.
  sqfliteFfiInit();
  databaseFactory = databaseFactoryFfi;

  try {
    // Open the existing EPMS database.
    await DatabaseHelper.instance.database;

    debugPrint('========================================');
    debugPrint('EPMS');
    debugPrint('========================================');
    debugPrint('Database connected successfully.');

    final topicCount =
        await DatabaseHelper.instance.getSyllabusTopicCount();

    debugPrint('Syllabus topics in database: $topicCount');
    debugPrint('========================================');

    runApp(
      EPMSApp(
        topicCount: topicCount,
      ),
    );
  } catch (e, stackTrace) {
    debugPrint('EPMS STARTUP ERROR: $e');
    debugPrint('$stackTrace');

    runApp(
      EPMSApp(
        error: e.toString(),
      ),
    );
  }
}

class EPMSApp extends StatelessWidget {
  final int? topicCount;
  final String? error;

  const EPMSApp({
    super.key,
    this.topicCount,
    this.error,
  });

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      title: 'EPMS',
      theme: ThemeData(
        useMaterial3: true,
      ),
      home: HomePage(
        topicCount: topicCount,
        error: error,
      ),
    );
  }
}

class HomePage extends StatelessWidget {
  final int? topicCount;
  final String? error;

  const HomePage({
    super.key,
    this.topicCount,
    this.error,
  });

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('EPMS'),
      ),
      body: Center(
        child: error != null
            ? Text(
                'Database error:\n\n$error',
                textAlign: TextAlign.center,
              )
            : Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  const Text(
                    'EPMS Home',
                    style: TextStyle(
                      fontSize: 32,
                      fontWeight: FontWeight.bold,
                    ),
                  ),
                  const SizedBox(height: 20),
                  Text(
                    'Syllabus topics loaded: $topicCount',
                    style: const TextStyle(
                      fontSize: 18,
                    ),
                  ),
                  const SizedBox(height: 40),
                  const Text(
                    'Select a subject to begin',
                    style: TextStyle(
                      fontSize: 20,
                    ),
                  ),
                ],
              ),
      ),
    );
  }
}
